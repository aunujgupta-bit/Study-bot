            async function generateStudyImage() {
                const input = document.getElementById("chat-text-input");
                const prompt = String(input && input.value || "").trim();
                if (!prompt) { toast("Pehle image ka description type karo"); if (input) input.focus(); return; }
                if (input) input.value = "";
                togglePlusTray(false);
                appendChatMessage("user", `
🎨 Generate an image: ${esc(prompt)}
`);
                const loading = appendChatMessage("bot", '
● AI image is being generated...
');
                const controller = new AbortController();
                const timeout = setTimeout(() => controller.abort(), 180000);
                try {
                    const response = await fetch(STUDY_BOT_API.replace(/\/$/, "") + "/generate-image", {
                        method: "POST",
                        headers: { "Content-Type": "application/json" },
                        body: JSON.stringify({ prompt }),
                        signal: controller.signal
                    });
                    const data = await response.json();
                    if (!response.ok || !data.success || !data.image) throw new Error(data.error || `Image request failed (${response.status})`);
                    const img = document.createElement("img");
                    img.src = String(data.image).startsWith("data:")
                        ? data.image
                        : `data:${data.mimeType || "image/jpeg"};base64,${data.image}`;
                    img.alt = prompt;
                    img.loading = "lazy";
                    img.className = "w-full rounded-xl border border-slate-200 mt-2";
                    img.style.maxHeight = "420px";
                    img.style.objectFit = "contain";
                    loading.innerHTML = `
🎨 Generated Image
${esc(prompt)}
`;
                    loading.appendChild(img);
                    const download = document.createElement("a");
                    download.href = img.src;
                    download.download = "Study-Bot-AI-Image.png";
                    download.textContent = "Download image";
                    download.className = "inline-block mt-2 px-3 py-2 rounded-lg bg-fuchsia-600 text-white text-xs font-bold";
                    loading.appendChild(download);
                    toast("AI image generated");
                } catch (error) {
                    loading.textContent = "Image generation failed: " + (error.name === "AbortError" ? "Request timed out" : error.message);
                    toast("Check Worker image-generation setup");
                } finally { clearTimeout(timeout); }
            }
